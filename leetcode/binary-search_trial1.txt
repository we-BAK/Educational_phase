class Solution:
    def search(self, nums: List[int], target: int) -> int:
        def s(left, right):
            if left > right:
                return -1
            mid = (left + right) // 2
            if nums[mid] == target:
                return mid
            elif nums[mid] < target:
                return s(mid + 1, right)
            else:
                return s(left, mid - 1)
        return s(0, len(nums) - 1)
