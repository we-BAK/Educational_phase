class Solution:
    def nextGreaterElement(self, nums1: List[int], nums2: List[int]) -> List[int]:
        res=defaultdict(lambda:-1)
        stack=[]
        for num in nums2:
            while stack and num >stack[-1]:
                res[stack.pop()]=num
            stack.append(num)
        answer=[res[num]for num in nums1]
        return answer
